## Why Event-Driven?

In microservices architectures, synchronous communication (REST/gRPC) creates temporal coupling: if one service is unavailable, the entire chain fails. Event-driven architectures solve this through asynchronous event-based communication.

## Fundamental Patterns

### 1. Event Notification

The simplest pattern: publish events to notify other services of changes.

```typescript
// Orders service publishes event
interface OrderCreatedEvent {
  eventType: 'OrderCreated';
  timestamp: Date;
  data: {
    orderId: string;
    customerId: string;
    items: Array<{ productId: string; quantity: number }>;
    totalAmount: number;
  };
}

// Producer
async function createOrder(order: Order): Promise<void> {
  // Persist the order
  await orderRepository.save(order);

  // Publish event
  await eventBus.publish('orders', {
    eventType: 'OrderCreated',
    timestamp: new Date(),
    data: {
      orderId: order.id,
      customerId: order.customerId,
      items: order.items,
      totalAmount: order.totalAmount,
    },
  });
}

// Consumer (Inventory Service)
eventBus.subscribe('orders', async (event: OrderCreatedEvent) => {
  if (event.eventType === 'OrderCreated') {
    for (const item of event.data.items) {
      await inventoryService.reserveStock(item.productId, item.quantity);
    }
  }
});
```

### 2. Event-Carried State Transfer

Events carry enough data so consumers don't need to fetch more information.

```typescript
// ❌ Poor event - forces consumer to fetch data
interface OrderCreatedPoor {
  orderId: string;  // Consumer needs to call Orders API
}

// ✅ Rich event - contains everything consumers need
interface OrderCreatedRich {
  eventType: 'OrderCreated';
  timestamp: Date;
  data: {
    orderId: string;
    customerId: string;
    customerEmail: string;  // For notification service
    customerName: string;
    shippingAddress: Address;  // For shipping service
    items: Array<{
      productId: string;
      productName: string;
      quantity: number;
      unitPrice: number;
    }>;
    totalAmount: number;
    paymentMethod: string;
  };
}
```

<Callout type="info" title="Trade-off">
  Rich events reduce inter-service calls, but increase message size and may expose more data than necessary.
</Callout>

### 3. Event Sourcing

Instead of storing only the current state, store all events that led to that state.

```typescript
// Event Store for Bank Account
interface AccountEvent {
  eventId: string;
  accountId: string;
  eventType: string;
  timestamp: Date;
  data: unknown;
}

interface AccountOpened {
  eventType: 'AccountOpened';
  data: { initialBalance: number; ownerName: string };
}

interface MoneyDeposited {
  eventType: 'MoneyDeposited';
  data: { amount: number; description: string };
}

interface MoneyWithdrawn {
  eventType: 'MoneyWithdrawn';
  data: { amount: number; description: string };
}

// Rebuild state from events
function rebuildAccount(events: AccountEvent[]): Account {
  return events.reduce((account, event) => {
    switch (event.eventType) {
      case 'AccountOpened':
        return {
          id: event.accountId,
          balance: event.data.initialBalance,
          ownerName: event.data.ownerName,
          isOpen: true,
        };
      case 'MoneyDeposited':
        return { ...account, balance: account.balance + event.data.amount };
      case 'MoneyWithdrawn':
        return { ...account, balance: account.balance - event.data.amount };
      case 'AccountClosed':
        return { ...account, isOpen: false };
      default:
        return account;
    }
  }, {} as Account);
}

// Persist new event
async function deposit(accountId: string, amount: number): Promise<void> {
  const events = await eventStore.getEvents(accountId);
  const account = rebuildAccount(events);

  if (!account.isOpen) {
    throw new Error('Account is closed');
  }

  await eventStore.append(accountId, {
    eventType: 'MoneyDeposited',
    data: { amount, description: 'Deposit' },
  });
}
```

### 4. CQRS (Command Query Responsibility Segregation)

Separates the write model (commands) from the read model (queries).

```typescript
// Write Side - Rich Domain Model
class Order {
  private events: DomainEvent[] = [];

  static create(customerId: string, items: OrderItem[]): Order {
    const order = new Order();
    order.apply(new OrderCreated(customerId, items));
    return order;
  }

  addItem(productId: string, quantity: number): void {
    this.apply(new ItemAdded(this.id, productId, quantity));
  }

  private apply(event: DomainEvent): void {
    this.events.push(event);
    this.when(event);
  }

  private when(event: DomainEvent): void {
    // Update internal state
  }
}

// Read Side - Projection optimized for queries
interface OrderReadModel {
  orderId: string;
  customerName: string;
  customerEmail: string;
  status: string;
  totalAmount: number;
  itemCount: number;
  createdAt: Date;
  // Denormalized data to avoid joins
}

class OrderProjection {
  async handle(event: DomainEvent): Promise<void> {
    switch (event.type) {
      case 'OrderCreated':
        await this.db.insert('orders_read', {
          orderId: event.orderId,
          customerName: event.customerName,
          status: 'created',
          // ...
        });
        break;
      case 'OrderShipped':
        await this.db.update('orders_read', {
          orderId: event.orderId,
          status: 'shipped',
          shippedAt: event.timestamp,
        });
        break;
    }
  }
}
```

## Saga Pattern for Distributed Transactions

When an operation involves multiple services, use Sagas to coordinate.

### Choreography-based Saga

<PlantUML
  code={`

skinparam participantBorderColor #5B8DD9

participant "Order\\nService" as order

  If any step fails, compensating events are emitted
  to rollback previous operations
end note

@enduml
  `}

  caption="Each service reacts to events and can emit compensations if something fails"
/>

```typescript
// Each service reacts to events and can emit compensations
class PaymentService {
  @EventHandler('OrderCreated')
  async handleOrderCreated(event: OrderCreatedEvent): Promise<void> {
    try {
      await this.processPayment(event.data.orderId, event.data.totalAmount);
      await this.eventBus.publish('payments', {
        eventType: 'PaymentCompleted',
        data: { orderId: event.data.orderId },
      });
    } catch (error) {
      await this.eventBus.publish('payments', {
        eventType: 'PaymentFailed',
        data: { orderId: event.data.orderId, reason: error.message },
      });
    }
  }
}

class OrderService {
  @EventHandler('PaymentFailed')
  async handlePaymentFailed(event: PaymentFailedEvent): Promise<void> {
    // Compensation: cancel the order
    await this.cancelOrder(event.data.orderId);
    await this.eventBus.publish('orders', {
      eventType: 'OrderCancelled',
      data: { orderId: event.data.orderId, reason: 'Payment failed' },
    });
  }
}
```

### Orchestration-based Saga

```typescript
// Saga Orchestrator centralizes the logic
class OrderSagaOrchestrator {
  async execute(orderId: string): Promise<void> {
    const saga = new Saga(orderId);

    try {
      // Step 1: Reserve inventory
      saga.addStep({
        execute: () => this.inventoryService.reserve(orderId),
        compensate: () => this.inventoryService.release(orderId),
      });

      // Step 2: Process payment
      saga.addStep({
        execute: () => this.paymentService.charge(orderId),
        compensate: () => this.paymentService.refund(orderId),
      });

      // Step 3: Create shipment
      saga.addStep({
        execute: () => this.shippingService.createShipment(orderId),
        compensate: () => this.shippingService.cancelShipment(orderId),
      });

      await saga.run();
    } catch (error) {
      await saga.compensate();
      throw error;
    }
  }
}
```

<Callout type="warning" title="Choosing the Approach">
  **Choreography**: More decoupled, better for simple flows. **Orchestration**: Easier to understand and debug, better for complex flows.
</Callout>

### Apache Kafka

```yaml
# docker-compose.yaml
services:
  kafka:
    image: confluentinc/cp-kafka:7.5.0
    environment:
      KAFKA_BROKER_ID: 1
      KAFKA_ZOOKEEPER_CONNECT: zookeeper:2181
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://kafka:9092
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
    ports:
      - "9092:9092"
```

```typescript
// Producer
import { Kafka } from 'kafkajs';

const kafka = new Kafka({ brokers: ['kafka:9092'] });
const producer = kafka.producer();

await producer.connect();
await producer.send({
  topic: 'orders',
  messages: [
    {
      key: orderId,  // Ensures ordering by order
      value: JSON.stringify(event),
      headers: {
        'event-type': 'OrderCreated',
        'correlation-id': correlationId,
      },
    },
  ],
});

// Consumer
const consumer = kafka.consumer({ groupId: 'inventory-service' });
await consumer.connect();
await consumer.subscribe({ topic: 'orders', fromBeginning: false });

await consumer.run({
  eachMessage: async ({ topic, partition, message }) => {
    const event = JSON.parse(message.value.toString());
    await processEvent(event);
  },
});
```
